#  GameDiffculty

class in mod.common.minecraftEnum

- 描述

	Peaceful = 0	# 和平
	Easy = 1		# 简单
	Normal = 2		# 普通
	Hard = 3		# 困难
	Count = 4		# 游戏难度种数
	Unknown = 5		# 未知
